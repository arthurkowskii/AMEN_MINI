cmake_minimum_required(VERSION 3.16)
project(amen_streams LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(amen_engine src/engine/pcm_wav.cpp src/engine/stream_mixer.cpp)
target_include_directories(amen_engine PUBLIC src/engine)
add_executable(stream_test tests/stream_test.cpp)
target_link_libraries(stream_test PRIVATE amen_engine)
add_executable(stream_io_test tests/stream_io_test.cpp)
target_include_directories(stream_io_test PRIVATE tests/stubs)
target_link_libraries(stream_io_test PRIVATE amen_engine)
if(MSVC)
    target_compile_options(amen_engine PRIVATE /W4 /WX)
    target_compile_options(stream_test PRIVATE /W4 /WX)
    target_compile_options(stream_io_test PRIVATE /W4 /WX)
else()
    target_compile_options(amen_engine PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(stream_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(stream_io_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME stream_test COMMAND stream_test)
add_test(NAME stream_io_test COMMAND stream_io_test)
