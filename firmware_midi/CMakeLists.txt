cmake_minimum_required(VERSION 3.16)
project(amen_midi LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(amen_midi_tests tests/test_main.cpp)
add_executable(amen_midi_front_panel_tests tests/test_front_panel.cpp)
add_executable(amen_midi_orientation_tests tests/test_orientation.cpp)
foreach(target amen_midi_tests amen_midi_front_panel_tests amen_midi_orientation_tests)
  target_include_directories(${target} PRIVATE teensy/amen_midi)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()

enable_testing()
add_test(NAME amen_midi_tests COMMAND amen_midi_tests)
add_test(NAME amen_midi_front_panel_tests COMMAND amen_midi_front_panel_tests)
add_test(NAME amen_midi_orientation_tests COMMAND amen_midi_orientation_tests)
